import "jsr:@supabase/functions-js/edge-runtime.d.ts"; import { withSupabase } from "npm:@supabase/server@1"; const DAILY_API_KEY = Deno.env.get("DAILY_API_KEY"); const ALLOWED_ROOMS = ["WKOLA", "PacaniSlovo"]; export default { fetch: withSupabase( { auth: "user" }, async (req, ctx) => { try { if (req.method !== "POST") { return Response.json( { error: "Only POST is allowed" }, { status: 405 } ); } if (!DAILY_API_KEY) { console.error("DAILY_API_KEY is missing"); return Response.json( { error: "Daily API key is not configured" }, { status: 500 } ); } // FIX: Supabase user ID is in .id, not .sub const userId = ctx.userClaims?.id; if (!userId) { console.error("Authenticated user ID is missing"); return Response.json( { error: "User not authenticated" }, { status: 401 } ); } let body; try { body = await req.json(); } catch { return Response.json( { error: "Invalid JSON request" }, { status: 400 } ); } const room = body?.room; if ( typeof room !== "string" || !ALLOWED_ROOMS.includes(room) ) { return Response.json( { error: "Unknown voice room" }, { status: 400 } ); } const { data: profile, error: profileError } = await ctx.supabase .from("profiles") .select("username") .eq("id", userId) .single(); if (profileError || !profile) { console.error("Profile error:", profileError); return Response.json( { error: "Profile not found" }, { status: 404 } ); } const expiration = Math.floor(Date.now() / 1000) + 3600; const dailyResponse = await fetch( "https://api.daily.co/v1/meeting-tokens", { method: "POST", headers: { Authorization: `Bearer ${DAILY_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ properties: { room_name: room, user_name: profile.username, user_id: userId, exp: expiration, is_owner: false } }) } ); let result; try { result = await dailyResponse.json(); } catch { result = {}; } if (!dailyResponse.ok) { console.error("Daily API error:", { status: dailyResponse.status, response: result }); return Response.json( { error: "Daily could not create meeting token", dailyStatus: dailyResponse.status, details: result }, { status: 502 } ); } if (!result.token) { console.error("Daily returned no token:", result); return Response.json( { error: "Daily returned an empty token" }, { status: 502 } ); } return Response.json({ token: result.token, room, username: profile.username }); } catch (error) { console.error("Function error:", error); return Response.json( { error: "Internal server error" }, { status: 500 } ); } } ) };
